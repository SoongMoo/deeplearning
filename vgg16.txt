from tensorflow.keras.applications.vgg16 import VGG16, preprocess_input, decode_predictions
from tensorflow.keras.preprocessing import image
import numpy as np

# 1. 테스트할 이미지 불러오기 및 전처리 (VGG16 원본 크기인 224x224 필수)
img_path = 'dog.jpg' # 테스트할 이미지 경로
img = image.load_img(img_path, target_size=(224, 224))
x = image.img_to_array(img)
x = np.expand_dims(x, axis=0)
x = preprocess_input(x) # VGG16 전용 전처리

# 2. include_top=True로 설정하여 출력층까지 전부 가져옵니다.
model = VGG16(weights='imagenet', include_top=True)

# 3. 예측하기
predictions = model.predict(x)

# 4. 결과 해석 (가장 확률이 높은 상위 3개 항목 출력)
print('예측 결과:', decode_predictions(predictions, top=3)[0])
